Configuração Supabase
Como usar:
Crie um projeto no Supabase, copie a URL e a chave anon public em Project Settings → API. Depois execute o SQL abaixo no SQL Editor.
Ver SQL para criação das tabelas 
create extension if not exists "pgcrypto";

create table if not exists jur_app_config (
  id boolean primary key default true,
  app_name text default 'LexBoard',
  app_subtitle text default 'Gestão Jurídica Inteligente',
  tipos jsonb default '[{"id":"demanda","label":"Demanda jurídica"},{"id":"tarefa","label":"Tarefa"}]',
  areas jsonb default '[{"id":"contratos","label":"Contratos"},{"id":"trabalhista","label":"Trabalhista"},{"id":"civel","label":"Cível"},{"id":"tributario","label":"Tributário"},{"id":"societario","label":"Societário"},{"id":"compliance","label":"Compliance"},{"id":"lgpd","label":"LGPD"},{"id":"administrativo","label":"Administrativo"}]',
  status jsonb default '[{"id":"triagem","label":"Triagem","color":"#556080"},{"id":"em_analise","label":"Em análise","color":"#4a80f5"},{"id":"em_execucao","label":"Em execução","color":"#f5a623"},{"id":"aguardando","label":"Aguardando terceiro","color":"#9b6bff"},{"id":"concluido","label":"Concluído","color":"#2ecc87"}]',
  prioridades jsonb default '[{"id":"baixa","label":"Baixa"},{"id":"media","label":"Média"},{"id":"alta","label":"Alta"},{"id":"critico","label":"Crítica"}]',
  riscos jsonb default '[{"id":"baixo","label":"Baixo"},{"id":"medio","label":"Médio"},{"id":"alto","label":"Alto"},{"id":"critico","label":"Crítico"}]',
  campos_extras jsonb default '[]',
  updated_at timestamptz default now()
);

insert into jur_app_config (id)
values (true)
on conflict (id) do nothing;

create table if not exists jur_profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  nome text,
  email text unique,
  perfil text not null default 'usuario',
  cargo text,
  cor text default '#4a80f5',
  ativo boolean default true,
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

create table if not exists jur_responsaveis (
  id uuid primary key default gen_random_uuid(),
  nome text not null,
  email text,
  cargo text,
  perfil text default 'usuario',
  cor text default '#4a80f5',
  ativo boolean default true,
  alertas_email boolean default true,
  alerta_status boolean default true,
  alerta_prazo boolean default true,
  alerta_atrib boolean default true,
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

create table if not exists jur_demandas (
  id uuid primary key default gen_random_uuid(),
  codigo text,
  tipo_registro text default 'tarefa',
  titulo text not null,
  descricao text,
  area_juridica text,
  risco text default 'medio',
  prioridade text default 'media',
  status text default 'triagem',
  responsavel_id uuid,
  prazo date,
  numero_processo text,
  parte_contraria text,
  valor_envolvido numeric,
  link_documento text,
  checklist jsonb default '[]',
  historico jsonb default '[]',
  extras jsonb default '{}',
  created_by uuid references auth.users(id) on delete set null,
  updated_by uuid references auth.users(id) on delete set null,
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

alter table jur_app_config enable row level security;
alter table jur_profiles enable row level security;
alter table jur_responsaveis enable row level security;
alter table jur_demandas enable row level security;

drop policy if exists read_app_config on jur_app_config;
create policy read_app_config on jur_app_config for select using (true);

drop policy if exists admin_write_app_config on jur_app_config;
create policy admin_write_app_config on jur_app_config
for all using (
  exists (
    select 1 from jur_profiles p
    where p.id = auth.uid()
    and p.perfil = 'admin'
    and p.ativo = true
  )
)
with check (
  exists (
    select 1 from jur_profiles p
    where p.id = auth.uid()
    and p.perfil = 'admin'
    and p.ativo = true
  )
);

drop policy if exists read_profiles on jur_profiles;
create policy read_profiles on jur_profiles for select using (auth.uid() is not null);

drop policy if exists self_insert_profile on jur_profiles;
create policy self_insert_profile on jur_profiles
for insert with check (id = auth.uid());

drop policy if exists self_or_admin_update_profile on jur_profiles;
create policy self_or_admin_update_profile on jur_profiles
for update using (
  id = auth.uid()
  or exists (
    select 1 from jur_profiles p
    where p.id = auth.uid()
    and p.perfil = 'admin'
    and p.ativo = true
  )
)
with check (
  id = auth.uid()
  or exists (
    select 1 from jur_profiles p
    where p.id = auth.uid()
    and p.perfil = 'admin'
    and p.ativo = true
  )
);

drop policy if exists read_responsaveis on jur_responsaveis;
create policy read_responsaveis on jur_responsaveis for select using (auth.uid() is not null);

drop policy if exists write_responsaveis_admin_gestor on jur_responsaveis;
create policy write_responsaveis_admin_gestor on jur_responsaveis
for all using (
  exists (
    select 1 from jur_profiles p
    where p.id = auth.uid()
    and p.perfil in ('admin','gestor')
    and p.ativo = true
  )
)
with check (
  exists (
    select 1 from jur_profiles p
    where p.id = auth.uid()
    and p.perfil in ('admin','gestor')
    and p.ativo = true
  )
);

drop policy if exists read_demandas on jur_demandas;
create policy read_demandas on jur_demandas for select using (auth.uid() is not null);

drop policy if exists write_demandas on jur_demandas;
create policy write_demandas on jur_demandas
for all using (
  exists (
    select 1 from jur_profiles p
    where p.id = auth.uid()
    and p.perfil in ('admin','gestor','advogado','usuario')
    and p.ativo = true
  )
)
with check (
  exists (
    select 1 from jur_profiles p
    where p.id = auth.uid()
    and p.perfil in ('admin','gestor','advogado','usuario')
    and p.ativo = true
  )
);
